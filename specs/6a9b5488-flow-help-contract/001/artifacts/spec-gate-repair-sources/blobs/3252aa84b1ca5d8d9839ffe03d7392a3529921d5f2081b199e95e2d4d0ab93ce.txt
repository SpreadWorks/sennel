import { createHash } from "node:crypto";
import { WorkerArtifactHandoffError } from "./worker-artifact-handoff-error.js";
import { SpecJsonValidator } from "../../lib/spec-json-validator.js";
import { compareText } from "./text-order.js";
import { RangedTextPromptElement } from "../../lib/prompt-batching.js";
import { workerArtifactStableStringify, MAX_WORKER_ARTIFACT_INPUT_BYTES } from "./worker-artifact-input-format.js";

const sourceDigest = (content) => createHash("sha256").update(content, "utf8").digest("hex");

/** Read-only captured evidence. Availability never implies investigation of other paths. */
export class SpecGateRepairSource {
  constructor({ id, origin, revision, content, availability = "available", required = !id?.startsWith("source:"), appliesTo = [] }) {
    if (typeof id !== "string" || !id || typeof origin !== "string" || !origin
      || typeof revision !== "string" || !revision || typeof content !== "string"
      || !["available", "missing", "unavailable"].includes(availability)
      || typeof required !== "boolean" || !Array.isArray(appliesTo)
      || appliesTo.some((scope) => typeof scope !== "string")) {
      throw new TypeError("Repair source requires an identity, origin, revision and text");
    }
    if (availability !== "available" && content !== "") throw new TypeError("Unavailable repair source cannot contain captured text");
    this.id = `evidence:${id}`;
    this.origin = origin;
    this.revision = revision;
    this.content = content;
    this.availability = availability;
    this.required = required;
    this.appliesTo = Object.freeze([...new Set(appliesTo)].sort());
    this.digest = sourceDigest(content);
    this.byteLength = Buffer.byteLength(content, "utf8");
    Object.freeze(this);
  }
  get isProjectRule() { return this.appliesTo.length > 0; }
  static referencedOrigins(value, origins) {
    // Match literal known paths in any language. A longer registered path owns
    // its complete occurrence, so its suffix cannot select a sibling source.
    if (origins.length === 0) return new Set();
    const paths = [...new Set(origins)];
    const referenced = new Set();
    const collect = (entry) => {
      if (typeof entry === "string") {
        const occurrences = new Map();
        for (const origin of paths) {
          for (let start = entry.indexOf(origin); start !== -1; start = entry.indexOf(origin, start + 1)) {
            if ((occurrences.get(start)?.length ?? 0) < origin.length) occurrences.set(start, origin);
          }
        }
        let end = 0;
        for (const start of [...occurrences.keys()].sort((left, right) => left - right)) {
          if (start < end) continue;
          const origin = occurrences.get(start);
          referenced.add(origin);
          end = start + origin.length;
        }
      } else if (entry !== null && typeof entry === "object") Object.values(entry).forEach(collect);
    };
    collect(value);
    return referenced;
  }
  appliesToOrigin(origin) {
    return this.appliesTo.some((scope) => scope === "." || origin.startsWith(`${scope}/`));
  }
  static assertAvailability(origin, availability, sourceId) {
    if (availability !== "available") {
      throw new WorkerArtifactHandoffError("invalid", "FLOW_SPEC_GATE_REPAIR_CONTEXT_UNAVAILABLE",
        "Selected Spec Gate repair context is unavailable", { data: { failureKind: "step-admission", origin,
          ...(sourceId === undefined ? {} : { sourceId }), ...(availability === undefined ? {} : { availability }) } });
    }
  }
  assertAvailable() { SpecGateRepairSource.assertAvailability(this.origin, this.availability, this.id); }
  query(query, binding, priorResults = []) {
    return SpecGateRepairSourceQueryResult.capture(this, query, binding, priorResults);
  }
  descriptor() {
    return { id: this.id, origin: this.origin, revision: this.revision, digest: this.digest,
      byteLength: this.byteLength, availability: this.availability, required: this.required, appliesTo: this.appliesTo };
  }
  toJSON() {
    return { origin: this.origin, revision: this.revision, content: this.content,
      snapshotId: this.id, snapshotDigest: this.digest, snapshotByteLength: this.byteLength, byteStart: 0, byteEnd: this.byteLength,
      sliceDigest: this.digest, availability: this.availability, appliesTo: this.appliesTo };
  }
}

/** Parent-owned full source capture, separate from the selected worker input. */
export class SpecGateRepairSourceSnapshots {
  #sources;
  constructor(sources) {
    if (!Array.isArray(sources) || sources.some((source) => !(source instanceof SpecGateRepairSource))
      || new Set(sources.map((source) => source.id)).size !== sources.length) {
      throw new TypeError("Repair snapshots require unique typed sources");
    }
    this.#sources = Object.freeze([...sources].sort((a, b) => compareText(a.id, b.id)));
    Object.freeze(this);
  }
  toJSON() { return { version: 1, sources: this.#sources.map((source) => ({ ...source.descriptor(), content: source.content })) }; }
  sources() { return [...this.#sources]; }
  static fromJSON(value) {
    if (!value || value.version !== 1 || Object.keys(value).sort().join(",") !== "sources,version"
      || !Array.isArray(value.sources)) throw new TypeError("Repair source snapshot format unavailable");
    return new SpecGateRepairSourceSnapshots(value.sources.map((entry) => {
      if (!entry.id?.startsWith("evidence:") || Object.keys(entry).sort().join(",")
        !== "appliesTo,availability,byteLength,content,digest,id,origin,required,revision") {
        throw new TypeError("Invalid repair source snapshot descriptor");
      }
      const source = new SpecGateRepairSource({ ...entry, id: entry.id.slice("evidence:".length) });
      if (source.digest !== entry.digest || source.byteLength !== entry.byteLength) throw new Error("Repair source snapshot digest mismatch");
      return source;
    }));
  }
}

/** Exact content-addressed catalog manifest, independent of an Attempt. */
export class SpecGateRepairSourceSnapshotReference {
  constructor(value) {
    if (!value || Object.keys(value).sort().join(",") !== "byteLength,digest"
      || !/^[a-f0-9]{64}$/.test(value.digest)
      || !Number.isSafeInteger(value.byteLength) || value.byteLength < 0) {
      throw new TypeError("Repair snapshot reference requires an exact digest and byte length");
    }
    this.digest = value.digest;
    this.byteLength = value.byteLength;
    Object.freeze(this);
  }
  toJSON() { return { digest: this.digest, byteLength: this.byteLength }; }
}

/** Bodyless source identities; available bodies are addressed by their digest. */
export class SpecGateRepairSourceSnapshotManifest {
  #descriptors;
  constructor(descriptors) {
    if (!Array.isArray(descriptors)) throw new TypeError("Repair manifest requires source descriptors");
    const seen = new Set();
    this.#descriptors = Object.freeze(descriptors.map((entry) => {
      if (!entry || Object.keys(entry).sort().join(",")
        !== "appliesTo,availability,byteLength,digest,id,origin,required,revision"
        || !entry.id?.startsWith("evidence:") || seen.has(entry.id)
        || !/^[a-f0-9]{64}$/.test(entry.digest)
        || !Number.isSafeInteger(entry.byteLength) || entry.byteLength < 0) {
        throw new TypeError("Repair manifest source descriptor is invalid");
      }
      const identity = new SpecGateRepairSource({ ...entry, id: entry.id.slice("evidence:".length), content: "" });
      if (entry.availability !== "available"
        && (entry.byteLength !== 0 || entry.digest !== identity.digest)
        || JSON.stringify(identity.appliesTo) !== JSON.stringify(entry.appliesTo)) {
        throw new TypeError("Repair manifest unavailable source cannot bind a body");
      }
      seen.add(entry.id);
      return Object.freeze({ ...entry, appliesTo: identity.appliesTo });
    }).sort((a, b) => compareText(a.id, b.id)));
    Object.freeze(this);
  }
  static fromSnapshots(snapshots) {
    if (!(snapshots instanceof SpecGateRepairSourceSnapshots)) throw new TypeError("Repair manifest requires typed snapshots");
    return new this(snapshots.sources().map((source) => source.descriptor()));
  }
  static fromJSON(value) {
    if (!value || value.version !== 1 || Object.keys(value).sort().join(",") !== "sources,version") {
      throw new TypeError("Repair source manifest format is unavailable");
    }
    const manifest = new this(value.sources);
    if (JSON.stringify(manifest.toJSON()) !== JSON.stringify(value)) throw new TypeError("Repair manifest must be canonically ordered");
    return manifest;
  }
  toJSON() { return { version: 1, sources: this.#descriptors.map((entry) => ({ ...entry })) }; }
  descriptors() { return [...this.#descriptors]; }
  bytes() { return Buffer.from(`${JSON.stringify(this.toJSON(), null, 2)}\n`, "utf8"); }
  reference() {
    const bytes = this.bytes();
    return new SpecGateRepairSourceSnapshotReference({ digest: createHash("sha256").update(bytes).digest("hex"), byteLength: bytes.length });
  }
}

/** Prepared offsets belong to the captured source, not to a caller's coordinate system. */
export class SpecGateRepairSourceTextIndex {
  #byteOffsets;
  #lineStarts;
  constructor(source) {
    if (!(source instanceof SpecGateRepairSource)) throw new TypeError("Source text index requires a captured source");
    this.sourceId = source.id; this.sourceDigest = source.digest;
    this.#byteOffsets = new Uint32Array(source.content.length + 1);
    this.#lineStarts = [0];
    let bytes = 0;
    for (let character = 0; character < source.content.length;) {
      const point = source.content.codePointAt(character);
      const width = point > 0xffff ? 2 : 1;
      bytes += point <= 0x7f ? 1 : point <= 0x7ff ? 2 : point <= 0xffff ? 3 : 4;
      if (width === 2) this.#byteOffsets[character + 1] = this.#byteOffsets[character];
      character += width;
      this.#byteOffsets[character] = bytes;
      if (point === 10) this.#lineStarts.push(character);
    }
    this.length = source.content.length;
    Object.freeze(this);
  }
  assertSource(source) {
    if (!(source instanceof SpecGateRepairSource) || source.id !== this.sourceId || source.digest !== this.sourceDigest) {
      throw new TypeError("Source text index belongs to a different captured source");
    }
  }
  #upperBound(values, target) {
    let low = 0; let high = values.length;
    while (low < high) { const middle = Math.floor((low + high) / 2);
      if (values[middle] <= target) low = middle + 1; else high = middle; }
    return low;
  }
  byteOffset(character) { return this.#byteOffsets[character]; }
  characterOffset(byte) {
    let position = this.#upperBound(this.#byteOffsets, byte) - 1;
    // A surrogate pair's interior shares its start offset; choose the code point start.
    if (position > 0 && this.#byteOffsets[position] === this.#byteOffsets[position - 1]) position--;
    return position;
  }
  lineWindow(start, end, beforeLines, afterLines) {
    const first = this.#upperBound(this.#lineStarts, start) - 1;
    const last = this.#upperBound(this.#lineStarts, end - 1) - 1;
    return { start: this.#lineStarts[Math.max(0, first - beforeLines)],
      end: this.#lineStarts[Math.min(this.#lineStarts.length, last + afterLines + 1)] ?? this.length };
  }
}

/** An exact literal inspection of captured bytes; cursors are authenticated receipts. */
export class SpecGateRepairSourceQuery {
  constructor(value) {
    if (!value || Object.keys(value).sort().join(",") !== "afterLines,beforeLines,cursor,literal,maxMatches,origin"
      || typeof value.origin !== "string" || !value.origin || typeof value.literal !== "string" || !value.literal || Buffer.from(value.literal, "utf8").toString("utf8") !== value.literal
      || !Number.isSafeInteger(value.beforeLines) || value.beforeLines < 0
      || !Number.isSafeInteger(value.afterLines) || value.afterLines < 0
      || !Number.isSafeInteger(value.maxMatches) || value.maxMatches < 1
      || value.cursor !== null && (typeof value.cursor !== "string" || !/^source-query:[a-f0-9]{64}$/.test(value.cursor))) {
      throw new TypeError("Source query requires an exact literal, line window, match bound and receipt cursor");
    }
    Object.assign(this, value);
    Object.freeze(this);
  }
  toJSON() { return { origin: this.origin, literal: this.literal, beforeLines: this.beforeLines,
    afterLines: this.afterLines, maxMatches: this.maxMatches, cursor: this.cursor }; }
  assertCursor(priorResults, { unitId, baseRevision }) {
    if (this.cursor === null) return null;
    const prior = priorResults.find((result) => result.id === this.cursor);
    const parameters = (query) => { const { cursor, ...value } = query.toJSON(); return workerArtifactStableStringify(value); };
    if (!(prior instanceof SpecGateRepairSourceQueryResult) || prior.nextCursor !== this.cursor
      || prior.unitId !== unitId || prior.baseRevision !== baseRevision
      || parameters(prior.query) !== parameters(this)) {
      throw new Error("Source query cursor is missing, complete or bound to different inspection input");
    }
    return prior;
  }
}

/** A complete inspection fact plus the exact immutable windows it selected. */
export class SpecGateRepairSourceQueryResult {
  static fromJSON(value) { return new this(value); }
  static #assertInputSize(actualBytes) {
    if (actualBytes > MAX_WORKER_ARTIFACT_INPUT_BYTES) {
      throw new WorkerArtifactHandoffError("invalid", "FLOW_SPEC_GATE_REPAIR_INPUT_TOO_LARGE",
        "Captured source inspection exceeds its immutable input file limit", { data: { failureKind: "step-admission",
          actualBytes, maximumBytes: MAX_WORKER_ARTIFACT_INPUT_BYTES } });
    }
  }
  constructor(value, ranges = []) {
    if (!value || Object.keys(value).sort().join(",") !== "baseRevision,complete,id,matches,nextCursor,query,searchedInterval,sourceByteLength,sourceDigest,sourceId,unitId,version"
      || value.version !== 1 || typeof value.unitId !== "string" || !value.unitId
      || !/^sha256:[a-f0-9]{64}$/.test(value.baseRevision) || !value.sourceId?.startsWith("evidence:source:")
      || !/^[a-f0-9]{64}$/.test(value.sourceDigest) || !Number.isSafeInteger(value.sourceByteLength) || value.sourceByteLength < 0
      || typeof value.complete !== "boolean" || !Array.isArray(value.matches)
      || !value.searchedInterval || Object.keys(value.searchedInterval).sort().join(",") !== "byteEnd,byteStart"
      || !Number.isSafeInteger(value.searchedInterval.byteStart) || !Number.isSafeInteger(value.searchedInterval.byteEnd)
      || value.searchedInterval.byteStart < 0 || value.searchedInterval.byteEnd < value.searchedInterval.byteStart
      || value.searchedInterval.byteEnd > value.sourceByteLength) {
      throw new TypeError("Invalid captured source query result");
    }
    this.query = new SpecGateRepairSourceQuery(value.query);
    if (value.matches.length > this.query.maxMatches
      || value.complete && value.searchedInterval.byteEnd !== value.sourceByteLength
      || !value.complete && (value.matches.length !== this.query.maxMatches || value.searchedInterval.byteEnd >= value.sourceByteLength)
      || value.matches.some((match, index) => !match
        || Object.keys(match).sort().join(",") !== "byteEnd,byteStart,rangeId"
        || !Number.isSafeInteger(match.byteStart) || !Number.isSafeInteger(match.byteEnd)
        || match.byteStart < value.searchedInterval.byteStart || match.byteEnd <= match.byteStart
        || match.byteEnd > value.sourceByteLength || match.byteStart >= value.searchedInterval.byteEnd
        || index > 0 && match.byteStart <= value.matches[index - 1].byteStart
        || typeof match.rangeId !== "string" || !match.rangeId.startsWith(`${value.sourceId}@bytes:`))) {
      throw new TypeError("Conflicting source query matches or search coverage");
    }
    const { id, nextCursor, ...unsigned } = value;
    const expectedId = `source-query:${sourceDigest(workerArtifactStableStringify(unsigned))}`;
    if (id !== expectedId || nextCursor !== (value.complete ? null : id)
      || !Array.isArray(ranges) || ranges.some((range) => !(range instanceof SpecGateRepairSourceRange))) {
      throw new TypeError("Conflicting source query result identity");
    }
    Object.assign(this, { ...value, query: this.query,
      searchedInterval: Object.freeze({ ...value.searchedInterval }),
      matches: Object.freeze(value.matches.map((match) => Object.freeze({ ...match }))), ranges: Object.freeze([...ranges]) });
    Object.freeze(this);
  }
  static capture(source, query, binding, priorResults = []) {
    if (!(source instanceof SpecGateRepairSource) || !(query instanceof SpecGateRepairSourceQuery)
      || source.required || source.isProjectRule || query.origin !== source.origin) {
      throw new TypeError("Source inspection requires an exact captured code source and typed query");
    }
    source.assertAvailable();
    const prior = query.assertCursor(priorResults, binding);
    if (prior && (prior.sourceId !== source.id || prior.sourceDigest !== source.digest
      || prior.sourceByteLength !== source.byteLength)) throw new Error("Source query cursor has a stale captured source");
    const startByte = prior?.searchedInterval.byteEnd ?? 0;
    const index = new SpecGateRepairSourceTextIndex(source);
    const start = index.characterOffset(startByte);
    const element = new RangedTextPromptElement({ id: source.id, sourceRevision: source.revision, sequence: 0, text: source.content });
    const ranges = new Map(); const windows = new Map(); const matches = [];
    let selectedBytes = Buffer.byteLength(workerArtifactStableStringify({ ...binding, query: query.toJSON(), sourceId: source.id,
      sourceDigest: source.digest, sourceByteLength: source.byteLength }), "utf8");
    let position = source.content.indexOf(query.literal, start);
    while (position !== -1 && matches.length < query.maxMatches) {
      const { start: left, end: right } = index.lineWindow(position, position + query.literal.length,
        query.beforeLines, query.afterLines);
      const windowKey = `${left}:${right}`;
      let range = windows.get(windowKey);
      if (range === undefined) {
        range = new SpecGateRepairSourceRange({ source, element: element.createRange({ start: left, end: right }), index });
        windows.set(windowKey, range);
      }
      if (!ranges.has(range.id)) {
        ranges.set(range.id, range);
        selectedBytes += Buffer.byteLength(workerArtifactStableStringify(range.value), "utf8");
      }
      const match = { byteStart: index.byteOffset(position), byteEnd: index.byteOffset(position + query.literal.length), rangeId: range.id };
      selectedBytes += Buffer.byteLength(workerArtifactStableStringify(match), "utf8");
      this.#assertInputSize(selectedBytes);
      matches.push(match);
      // Search overlapping matches at the next Unicode code point, never inside a surrogate pair.
      const next = position + (source.content.codePointAt(position) > 0xffff ? 2 : 1);
      position = source.content.indexOf(query.literal, next);
    }
    const unsigned = { version: 1, ...binding, sourceId: source.id, sourceDigest: source.digest,
      sourceByteLength: source.byteLength, query: query.toJSON(),
      searchedInterval: { byteStart: startByte, byteEnd: position === -1 ? source.byteLength
        : index.byteOffset(position) }, matches, complete: position === -1 };
    const id = `source-query:${sourceDigest(workerArtifactStableStringify(unsigned))}`;
    const result = new this({ ...unsigned, id, nextCursor: unsigned.complete ? null : id }, [...ranges.values()]);
    const serializedBytes = Buffer.byteLength(workerArtifactStableStringify(result.toJSON()), "utf8")
      + [...ranges.values()].reduce((total, range) => total + Buffer.byteLength(workerArtifactStableStringify(range.value), "utf8"), 0);
    this.#assertInputSize(serializedBytes);
    return result;
  }
  toJSON() { const { ranges, query, ...value } = this; return { ...value, query: query.toJSON() }; }
}

/** Exact UTF-8 coverage bound to one captured source. */
export class SpecGateRepairSourceRange {
  /** Available tracked code ranges use this origin prefix; rules and canonical artifacts do not. */
  static prefixForOrigin(origin) {
    if (typeof origin !== "string" || !origin) throw new TypeError("Repair source range prefix requires an origin");
    return `evidence:source:${origin}@bytes:`;
  }

  constructor({ source, element, index = null }) {
    if (!(source instanceof SpecGateRepairSource)) throw new TypeError("Repair source range requires a captured source");
    if (index !== null) {
      if (!(index instanceof SpecGateRepairSourceTextIndex)) throw new TypeError("Repair source offsets require a typed captured index");
      index.assertSource(source);
    }
    const byteStart = index === null ? Buffer.byteLength(source.content.slice(0, element.start), "utf8") : index.byteOffset(element.start);
    const byteEnd = index === null ? byteStart + Buffer.byteLength(element.text, "utf8") : index.byteOffset(element.end);
    this.value = Object.freeze({ ...source.toJSON(), content: element.text, byteStart, byteEnd,
      sliceDigest: sourceDigest(element.text) });
    this.id = `${source.id}@bytes:${byteStart}:${byteEnd}:${source.digest}`;
    this.digest = this.value.sliceDigest;
    Object.freeze(this);
  }
  static sameSnapshot(left, right) {
    return left?.snapshotId !== undefined && right?.snapshotId === left.snapshotId
      && right.snapshotDigest === left.snapshotDigest;
  }
  static covers(available, requested) {
    return this.sameSnapshot(available, requested)
      && available.byteStart <= requested.byteStart && available.byteEnd >= requested.byteEnd;
  }
  static uncoveredBytes(requested, available) {
    const intervals = available.filter((value) => this.sameSnapshot(value, requested))
      .map((value) => [Math.max(requested.byteStart, value.byteStart), Math.min(requested.byteEnd, value.byteEnd)])
      .filter(([start, end]) => end > start).sort(([left], [right]) => left - right);
    let overlap = 0; let frontier = requested.byteStart;
    for (const [start, end] of intervals) {
      overlap += Math.max(0, end - Math.max(frontier, start));
      frontier = Math.max(frontier, end);
    }
    return requested.byteEnd - requested.byteStart - overlap;
  }
  static uncoveredRangeBytes(requested, available) {
    const covered = [...available];
    return requested.reduce((bytes, value) => {
      const additional = this.uncoveredBytes(value, covered);
      covered.push(value);
      return bytes + additional;
    }, 0);
  }
  static validate(value, digest) {
    if (!value || Object.keys(value).sort().join(",") !== "appliesTo,availability,byteEnd,byteStart,content,origin,revision,sliceDigest,snapshotByteLength,snapshotDigest,snapshotId"
      || typeof value.origin !== "string" || !value.origin || typeof value.revision !== "string" || !value.revision
      || !value.snapshotId?.startsWith("evidence:") || !/^[a-f0-9]{64}$/.test(value.snapshotDigest)
      || !Number.isSafeInteger(value.byteStart) || !Number.isSafeInteger(value.byteEnd)
      || value.byteStart < 0 || value.byteEnd < value.byteStart || typeof value.content !== "string"
      || !Number.isSafeInteger(value.snapshotByteLength) || value.byteEnd > value.snapshotByteLength
      || !["available", "missing", "unavailable"].includes(value.availability)
      || !Array.isArray(value.appliesTo) || value.appliesTo.some((scope) => typeof scope !== "string")
      || (value.availability !== "available" && value.content !== "")
      || Buffer.byteLength(value.content, "utf8") !== value.byteEnd - value.byteStart
      || sourceDigest(value.content) !== value.sliceDigest || digest !== value.sliceDigest) {
      throw new Error("Conflicting repair bundle source digest or byte range");
    }
  }
}

/** One version-bound, permission-limited parent input for Gate repair. */
export class SpecGateRepairInput {
  constructor({ repair, spec, baseRevision, specByteLength, context, sourceDescriptor, review, attempt,
    observationIdentities, validator }) {
    if (!(validator instanceof SpecJsonValidator)) throw new TypeError("Gate repair input requires its canonical Spec validator");
    this.repair = repair;
    this.spec = Object.freeze(structuredClone(spec));
    this.baseRevision = baseRevision;
    this.specByteLength = specByteLength;
    this.context = context;
    this.sourceDescriptor = Object.freeze(structuredClone(sourceDescriptor));
    this.review = review;
    this.attempt = Object.freeze({ id: attempt.id, sequence: attempt.sequence });
    this.observationIdentities = Object.freeze(observationIdentities.map((identity) => Object.freeze(identity)));
    this.validator = validator;
    Object.freeze(this);
  }
}
